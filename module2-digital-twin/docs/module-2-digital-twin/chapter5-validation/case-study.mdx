# Complete Case Study: Sim-to-Real Transfer

This comprehensive case study demonstrates the complete process of developing a humanoid robot navigation system that transfers successfully from simulation to reality. We'll walk through the entire pipeline from initial simulation development to real-world deployment and validation.

## Case Study Overview

### Project: Autonomous Navigation for Humanoid Robot

**Objective**: Develop a humanoid robot capable of navigating through a cluttered office environment to reach specified destinations while avoiding obstacles and interacting safely with humans.

**Environment**: Office setting with desks, chairs, doorways, and people moving around

**Robot**: Humanoid robot with wheeled base, manipulator arms, and sensor suite

**Timeline**: 6-month development cycle

### Initial Simulation Training

We began by training our navigation system in simulation using reinforcement learning:

```python
#!/usr/bin/env python3
"""
Simulation training for humanoid robot navigation
"""
import rospy
import numpy as np
from geometry_msgs.msg import Twist, Pose
from sensor_msgs.msg import LaserScan
from nav_msgs.msg import Odometry
from std_msgs.msg import Bool
import gym
from gym import spaces
import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque
import random

class NavigationEnv(gym.Env):
    def __init__(self):
        super(NavigationEnv, self).__init__()

        # Define action space: linear velocity and angular velocity
        self.action_space = spaces.Box(
            low=np.array([-0.5, -1.0]),  # min linear vel, min angular vel
            high=np.array([0.5, 1.0]),  # max linear vel, max angular vel
            dtype=np.float32
        )

        # Define observation space: LiDAR scan (720 points) + goal direction + robot velocity
        self.observation_space = spaces.Box(
            low=np.array([-1.0] * 720 + [-1, -1, -0.5, -1.0]),  # LiDAR + goal_dir + vel
            high=np.array([30.0] * 720 + [1, 1, 0.5, 1.0]),   # LiDAR + goal_dir + vel
            dtype=np.float32
        )

        # ROS subscribers and publishers
        self.lidar_sub = rospy.Subscriber('/scan', LaserScan, self.lidar_callback)
        self.odom_sub = rospy.Subscriber('/odom', Odometry, self.odom_callback)
        self.cmd_vel_pub = rospy.Publisher('/cmd_vel', Twist, queue_size=10)
        self.goal_pub = rospy.Publisher('/goal_reached', Bool, queue_size=10)

        # Internal state
        self.lidar_data = np.ones(720) * 10.0  # Initialize with 10m range
        self.robot_pose = Pose()
        self.robot_twist = Twist()
        self.goal_position = np.array([5.0, 5.0])  # Target position
        self.collision_threshold = 0.5  # Collision distance in meters

    def lidar_callback(self, msg):
        """Process LiDAR data"""
        if len(msg.ranges) == 720:
            self.lidar_data = np.array(msg.ranges)
            # Replace invalid values with max range
            self.lidar_data[np.isnan(self.lidar_data)] = 30.0
            self.lidar_data[self.lidar_data > 30.0] = 30.0

    def odom_callback(self, msg):
        """Process odometry data"""
        self.robot_pose = msg.pose.pose
        self.robot_twist = msg.twist.twist

    def step(self, action):
        """Execute action and return observation, reward, done, info"""
        # Send velocity command to robot
        cmd_vel = Twist()
        cmd_vel.linear.x = action[0]
        cmd_vel.angular.z = action[1]
        self.cmd_vel_pub.publish(cmd_vel)

        # Wait for next observation (simulate delay)
        rospy.sleep(0.1)

        # Calculate reward
        reward, done, info = self.calculate_reward_and_done()

        # Get current observation
        obs = self.get_observation()

        return obs, reward, done, info

    def get_observation(self):
        """Get current observation"""
        # Calculate goal direction relative to robot
        robot_pos = np.array([self.robot_pose.position.x, self.robot_pose.position.y])
        goal_dir = self.goal_position - robot_pos
        goal_distance = np.linalg.norm(goal_dir)

        if goal_distance > 0:
            goal_dir_normalized = goal_dir / goal_distance
        else:
            goal_dir_normalized = np.array([0.0, 0.0])

        # Normalize robot velocity
        linear_vel = self.robot_twist.linear.x
        angular_vel = self.robot_twist.angular.z

        # Combine all observations
        observation = np.concatenate([
            self.lidar_data,  # 720 LiDAR values
            goal_dir_normalized,  # 2 goal direction values
            [linear_vel, angular_vel]  # 2 velocity values
        ])

        return observation

    def calculate_reward_and_done(self):
        """Calculate reward and check if episode is done"""
        # Get current robot position
        robot_pos = np.array([self.robot_pose.position.x, self.robot_pose.position.y])

        # Calculate distance to goal
        distance_to_goal = np.linalg.norm(self.goal_position - robot_pos)

        # Calculate distance to nearest obstacle
        min_obstacle_dist = np.min(self.lidar_data)

        # Reward components
        goal_reward = -distance_to_goal  # Negative distance (closer is better)

        # Collision penalty
        collision_penalty = 0
        if min_obstacle_dist < self.collision_threshold:
            collision_penalty = -100

        # Progress reward (encourage movement toward goal)
        progress_reward = 0
        if hasattr(self, 'prev_distance_to_goal'):
            if distance_to_goal < self.prev_distance_to_goal:
                progress_reward = 1.0
            else:
                progress_reward = -0.5  # Penalty for moving away

        self.prev_distance_to_goal = distance_to_goal

        # Total reward
        reward = goal_reward * 0.1 + collision_penalty + progress_reward

        # Check termination conditions
        done = False
        info = {}

        if distance_to_goal < 0.5:  # Reached goal
            reward += 100  # Large bonus for reaching goal
            done = True
            info['success'] = True
        elif min_obstacle_dist < 0.3:  # Collision
            reward -= 50  # Penalty for collision
            done = True
            info['collision'] = True
        elif distance_to_goal > 20:  # Too far from goal
            done = True
            info['out_of_bounds'] = True

        return reward, done, info

    def reset(self):
        """Reset environment"""
        # In simulation, we might reset robot position
        # For this example, we'll just return initial observation
        self.prev_distance_to_goal = float('inf')
        return self.get_observation()

    def render(self, mode='human'):
        """Render environment (not implemented for this example)"""
        pass
```

### Deep Reinforcement Learning Implementation

We implemented a Deep Q-Network (DQN) for navigation:

```python
class DQN(nn.Module):
    def __init__(self, input_size, action_size):
        super(DQN, self).__init__()

        # Convolutional layers for processing LiDAR data
        self.conv_layers = nn.Sequential(
            nn.Conv1d(in_channels=1, out_channels=32, kernel_size=5, stride=2),
            nn.ReLU(),
            nn.Conv1d(in_channels=32, out_channels=64, kernel_size=5, stride=2),
            nn.ReLU(),
            nn.Conv1d(in_channels=64, out_channels=64, kernel_size=3, stride=1),
            nn.ReLU()
        )

        # Calculate conv output size
        conv_out_size = self._get_conv_out_size(input_size)

        # Fully connected layers
        self.fc_layers = nn.Sequential(
            nn.Linear(conv_out_size + 2, 512),  # +2 for goal direction
            nn.ReLU(),
            nn.Linear(512, 256),
            nn.ReLU(),
            nn.Linear(256, action_size)
        )

    def _get_conv_out_size(self, input_size):
        """Calculate output size of convolutional layers"""
        x = torch.zeros(1, 1, input_size)
        x = self.conv_layers(x)
        return int(np.prod(x.size()))

    def forward(self, state):
        # Split state into LiDAR data and other features
        lidar_data = state[:, :-2].unsqueeze(1)  # Add channel dimension
        other_features = state[:, -2:]  # Goal direction

        # Process LiDAR data through conv layers
        conv_out = self.conv_layers(lidar_data)
        conv_out = torch.flatten(conv_out, start_dim=1)

        # Concatenate with other features
        combined = torch.cat([conv_out, other_features], dim=1)

        # Process through FC layers
        q_values = self.fc_layers(combined)

        return q_values

class DQNAgent:
    def __init__(self, state_size, action_size, lr=1e-4):
        self.state_size = state_size
        self.action_size = action_size
        self.memory = deque(maxlen=10000)
        self.epsilon = 1.0  # Exploration rate
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
        self.learning_rate = lr
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        # Neural networks
        self.q_network = DQN(state_size, action_size).to(self.device)
        self.target_network = DQN(state_size, action_size).to(self.device)
        self.optimizer = optim.Adam(self.q_network.parameters(), lr=lr)

        # Update target network
        self.update_target_network()

    def remember(self, state, action, reward, next_state, done):
        """Store experience in replay memory"""
        self.memory.append((state, action, reward, next_state, done))

    def act(self, state):
        """Choose action using epsilon-greedy policy"""
        if np.random.random() <= self.epsilon:
            return np.random.choice(self.action_size)

        state_tensor = torch.FloatTensor(state).unsqueeze(0).to(self.device)
        q_values = self.q_network(state_tensor)
        return np.argmax(q_values.cpu().data.numpy())

    def replay(self, batch_size=32):
        """Train on batch of experiences"""
        if len(self.memory) < batch_size:
            return

        batch = random.sample(self.memory, batch_size)
        states = torch.FloatTensor([e[0] for e in batch]).to(self.device)
        actions = torch.LongTensor([e[1] for e in batch]).to(self.device)
        rewards = torch.FloatTensor([e[2] for e in batch]).to(self.device)
        next_states = torch.FloatTensor([e[3] for e in batch]).to(self.device)
        dones = torch.BoolTensor([e[4] for e in batch]).to(self.device)

        current_q_values = self.q_network(states).gather(1, actions.unsqueeze(1))
        next_q_values = self.target_network(next_states).max(1)[0].detach()
        target_q_values = rewards + (0.99 * next_q_values * ~dones)

        loss = nn.MSELoss()(current_q_values.squeeze(), target_q_values)

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay

    def update_target_network(self):
        """Copy weights from main network to target network"""
        self.target_network.load_state_dict(self.q_network.state_dict())
```

## Phase 3: Validation and Testing

### Simulation Performance Evaluation

We evaluated our trained agent in simulation with various metrics:

```python
class SimulationEvaluator:
    def __init__(self, agent, env):
        self.agent = agent
        self.env = env
        self.metrics = {
            'success_rate': [],
            'path_efficiency': [],
            'collision_rate': [],
            'average_time': [],
            'smoothness': []
        }

    def evaluate_agent(self, num_episodes=100):
        """Comprehensive evaluation of agent performance"""
        results = []

        for episode in range(num_episodes):
            state = self.env.reset()
            total_reward = 0
            steps = 0
            path_length = 0.0
            prev_position = None

            # Store trajectory for analysis
            trajectory = []

            for step in range(1000):  # Max steps
                # Use trained policy (no exploration)
                self.agent.epsilon = 0.0
                action_idx = self.agent.act(state)
                action = self.env.discretize_action(action_idx)

                next_state, reward, done, info = self.env.step(action)

                # Track trajectory
                current_pos = [self.env.robot_pose.position.x, self.env.robot_pose.position.y]
                trajectory.append(current_pos)

                # Calculate path length
                if prev_position is not None:
                    dist = np.linalg.norm(np.array(current_pos) - np.array(prev_position))
                    path_length += dist
                prev_position = current_pos

                state = next_state
                total_reward += reward
                steps += 1

                if done:
                    break

            # Calculate episode metrics
            success = info.get('success', False)
            collision = info.get('collision', False)

            # Calculate path efficiency (optimal distance / actual path length)
            start_pos = trajectory[0] if trajectory else [0, 0]
            goal_pos = self.env.goal_position
            optimal_distance = np.linalg.norm(np.array(goal_pos) - np.array(start_pos))
            path_efficiency = optimal_distance / path_length if path_length > 0 else 0

            # Calculate trajectory smoothness (inverse of total curvature)
            smoothness = self.calculate_trajectory_smoothness(trajectory)

            episode_result = {
                'episode': episode,
                'success': success,
                'collision': collision,
                'steps': steps,
                'total_reward': total_reward,
                'path_length': path_length,
                'path_efficiency': path_efficiency,
                'execution_time': steps * 0.1,  # Assuming 0.1s per step
                'smoothness': smoothness,
                'trajectory': trajectory
            }

            results.append(episode_result)

        # Calculate aggregate metrics
        self.calculate_aggregate_metrics(results)

        return results

    def calculate_trajectory_smoothness(self, trajectory):
        """Calculate trajectory smoothness based on curvature"""
        if len(trajectory) < 3:
            return 1.0  # Perfectly smooth for very short trajectories

        total_curvature = 0.0
        segments = 0

        for i in range(1, len(trajectory) - 1):
            p1 = np.array(trajectory[i-1])
            p2 = np.array(trajectory[i])
            p3 = np.array(trajectory[i+1])

            # Calculate angle between consecutive segments
            v1 = p2 - p1
            v2 = p3 - p2

            if np.linalg.norm(v1) > 0 and np.linalg.norm(v2) > 0:
                cos_angle = np.dot(v1, v2) / (np.linalg.norm(v1) * np.linalg.norm(v2))
                # Clamp to avoid numerical errors
                cos_angle = np.clip(cos_angle, -1.0, 1.0)
                angle = np.arccos(cos_angle)

                # Curvature is related to how sharp the turn is
                total_curvature += angle
                segments += 1

        # Smoothness is inverse of average curvature
        avg_curvature = total_curvature / segments if segments > 0 else 0
        smoothness = 1.0 / (1.0 + avg_curvature)  # Higher values = smoother

        return smoothness

    def calculate_aggregate_metrics(self, results):
        """Calculate aggregate performance metrics"""
        successful_episodes = [r for r in results if r['success']]
        collided_episodes = [r for r in results if r['collision']]

        # Success rate
        self.metrics['success_rate'] = len(successful_episodes) / len(results) if results else 0

        # Path efficiency (only for successful episodes)
        if successful_episodes:
            efficiencies = [ep['path_efficiency'] for ep in successful_episodes]
            self.metrics['path_efficiency'] = np.mean(efficiencies)
        else:
            self.metrics['path_efficiency'] = 0

        # Collision rate
        self.metrics['collision_rate'] = len(collided_episodes) / len(results) if results else 0

        # Average time (only for successful episodes)

            times = [ep['execution_time'] for ep in successful_episodes]
            self.metrics['average_time'] = np.mean(times)
        else:

        # Average smoothness (only for successful episodes)

        else:

    def generate_evaluation_report(self):
```

### Best Practices for Sim-to-Real Transfer

1. **Start Simple**: Begin with basic tasks and gradually increase complexity.

2. **Use Domain Randomization**: Systematically vary simulation parameters to improve robustness.

3. **Validate Early and Often**: Regular validation identifies issues before they become problematic.

4. **Maintain Safety Margins**: Always include safety buffers for real-world deployment.

5. **Collect Real Data**: Use real-world data to continuously improve simulation models.

6. **Plan for Iteration**: Expect multiple cycles of simulation-training-validation-refinement.
