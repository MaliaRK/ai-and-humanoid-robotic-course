# Isaac Sim Navigation Scene Setup

## Overview

This chapter covers the creation and configuration of Isaac Sim scenes specifically designed for humanoid robot navigation. These scenes include realistic environments with obstacles, human agents, and navigation goals to test and validate navigation algorithms for humanoid robots.

## Scene Architecture

### Core Components
The navigation scene consists of several key components:

1. **Environment**: Room or outdoor space with appropriate dimensions
2. **Humanoid Robot**: The navigation agent with proper physical properties
3. **Static Obstacles**: Fixed obstacles for navigation challenge
4. **Dynamic Elements**: Moving obstacles or human agents
5. **Navigation Goals**: Waypoints for navigation testing
6. **Sensors**: Cameras, IMU, LiDAR for perception
7. **Lighting**: Proper illumination for visual perception

### Scene Structure
```
/World
├── /Environment
│   ├── /Room (walls, floor, ceiling)
│   └── /Obstacles (static and dynamic)
├── /HumanoidRobot
│   ├── /Body (physical representation)
│   ├── /Sensors (cameras, IMU, etc.)
│   └── /Controllers (navigation interface)
├── /Humans
│   ├── /Human0
│   ├── /Human1
│   └── ...
├── /NavigationGoals
│   ├── /Goal0
│   ├── /Goal1
│   └── ...
└── /Lighting
    ├── /DomeLight
    └── /KeyLight
```

class DynamicHumanAgent:
    def __init__(self, prim_path, start_position, color):
        self.prim_path = prim_path
        self.current_position = start_position
        self.color = color
        self.movement_pattern = "stationary"  # Options: stationary, patrol, random_walk
        self.waypoints = []
        self.current_waypoint_index = 0
        self.movement_speed = 0.5  # m/s

    def create_in_stage(self):
        """Create the human agent in the USD stage"""
        # Create human body (torso)
        create_primitive(
            prim_path=f"{self.prim_path}/Body",
            primitive_type="Capsule",
            position=np.array(self.current_position),
            scale=np.array([0.15, 0.12, 0.8]),
            color=self.color
        )

        # Create human head
        create_primitive(
            prim_path=f"{self.prim_path}/Head",
            primitive_type="Sphere",
            position=np.array([self.current_position[0],
                              self.current_position[1],
                              self.current_position[2] + 0.8]),
            scale=np.array([0.1, 0.1, 0.1]),
            color=np.array([0.95, 0.9, 0.85])
        )

    def set_patrol_pattern(self, waypoints):
        """Set a patrol movement pattern"""
        self.movement_pattern = "patrol"
        self.waypoints = waypoints
        self.current_waypoint_index = 0

    def set_random_walk_pattern(self, area_center, area_radius):
        """Set a random walk movement pattern"""
        self.movement_pattern = "random_walk"
        self.area_center = area_center
        self.area_radius = area_radius

    def update_position(self, dt):
        """Update human position based on movement pattern"""
        if self.movement_pattern == "patrol" and self.waypoints:
            # Move toward current waypoint
            current_waypoint = self.waypoints[self.current_waypoint_index]
            direction = np.array(current_waypoint) - np.array(self.current_position)
            distance = np.linalg.norm(direction)

            if distance < 0.1:  # Reached waypoint
                self.current_waypoint_index = (self.current_waypoint_index + 1) % len(self.waypoints)
                current_waypoint = self.waypoints[self.current_waypoint_index]
                direction = np.array(current_waypoint) - np.array(self.current_position)

            # Move toward waypoint
            if distance > 0:
                direction = direction / distance  # Normalize
                movement = direction * self.movement_speed * dt
                self.current_position = self.current_position + movement

                # Update USD stage with new position
                self.update_usd_position()

        elif self.movement_pattern == "random_walk":
            # Move randomly within area
            angle = np.random.uniform(0, 2*np.pi)
            step = np.random.uniform(0, self.movement_speed * dt)
            dx = step * np.cos(angle)
            dy = step * np.sin(angle)

            new_x = self.current_position[0] + dx
            new_y = self.current_position[1] + dy

            # Stay within area bounds
            dist_from_center = np.sqrt((new_x - self.area_center[0])**2 + (new_y - self.area_center[1])**2)
            if dist_from_center <= self.area_radius:
                self.current_position[0] = new_x
                self.current_position[1] = new_y
                self.update_usd_position()

# Example: Navigation test scenarios for humanoid robot
class NavigationTestScenarios:
    """Test scenarios for humanoid robot navigation in Isaac Sim"""

    def __init__(self):
        self.scenarios = {
            "basic_navigation": self.test_basic_navigation,
            "obstacle_avoidance": self.test_obstacle_avoidance,
            "social_navigation": self.test_social_navigation,
            "dynamic_obstacle_avoidance": self.test_dynamic_obstacle_avoidance,
            "narrow_passage": self.test_narrow_passage_navigation,
            "multi_goal_navigation": self.test_multi_goal_navigation
        }

    def test_basic_navigation(self, robot, goals):
        """Test basic navigation to multiple goals"""
        print("Testing basic navigation...")

        results = []
        for i, goal in enumerate(goals):
            print(f"Navigating to goal {i+1}: {goal}")

            # Send navigation goal
            robot.navigate_to_pose(goal)

            # Wait for completion
            success, time_taken = robot.wait_for_navigation_completion(timeout=60.0)

            results.append({
                "goal": goal,
                "success": success,
                "time": time_taken,
                "path_length": robot.get_traveled_distance()
            })

            print(f"Goal {i+1} result: Success={success}, Time={time_taken:.2f}s")

        return results

    def test_obstacle_avoidance(self, robot, obstacle_positions):
        """Test obstacle avoidance capabilities"""
        print("Testing obstacle avoidance...")

        # Set up navigation with obstacles
        robot.set_costmap_obstacles(obstacle_positions)

        # Navigate through obstacle course
        goal = [3.0, 3.0, 0.0]
        robot.navigate_to_pose(goal)

        # Monitor for safe navigation
        success, time_taken = robot.wait_for_navigation_completion(timeout=120.0)

        # Check for collisions
        collisions = robot.get_collision_count()

        result = {
            "success": success,
            "time": time_taken,
            "collisions": collisions,
            "obstacle_positions": obstacle_positions
        }

        print(f"Obstacle avoidance test: Success={success}, Collisions={collisions}")
        return result

    def test_social_navigation(self, robot, human_agents):
        """Test navigation with human agents"""
        print("Testing social navigation...")

        # Start human agents moving
        for human in human_agents:
            human.start_movement()

        # Navigate while humans are moving
        goals = [[2.0, 2.0, 0.0], [-1.0, 3.0, 0.0], [3.0, -2.0, 0.0]]

        results = []
        for goal in goals:
            robot.navigate_to_pose(goal)
            success, time_taken = robot.wait_for_navigation_completion(timeout=60.0)

            # Check for social navigation compliance
            personal_space_violations = robot.get_personal_space_violations()

            results.append({
                "goal": goal,
                "success": success,
                "time": time_taken,
                "personal_space_violations": personal_space_violations
            })

        # Stop human agents
        for human in human_agents:
            human.stop_movement()

        print(f"Social navigation test completed")
        return results

    def test_dynamic_obstacle_avoidance(self, robot, dynamic_obstacles):
        """Test navigation with moving obstacles"""
        print("Testing dynamic obstacle avoidance...")

        # Start dynamic obstacles moving
        for obstacle in dynamic_obstacles:
            obstacle.start_movement()

        # Navigate through dynamic environment
        goal = [2.5, 2.5, 0.0]
        robot.navigate_to_pose(goal)

        success, time_taken = robot.wait_for_navigation_completion(timeout=90.0)

        # Check for collisions and successful avoidance
        collisions = robot.get_collision_count()
        path_efficiency = robot.get_path_efficiency()

        result = {
            "success": success,
            "time": time_taken,
            "collisions": collisions,
            "path_efficiency": path_efficiency,
            "dynamic_obstacles": len(dynamic_obstacles)
        }

        # Stop dynamic obstacles
        for obstacle in dynamic_obstacles:
            obstacle.stop_movement()

        print(f"Dynamic obstacle test: Success={success}, Collisions={collisions}")
        return result

    def run_all_tests(self, robot, scene_elements):
        """Run all navigation tests"""
        print("Running all navigation tests...")

        results = {}

        # Run each test scenario
        for scenario_name, test_func in self.scenarios.items():
            print(f"\n--- Running {scenario_name} ---")
            try:
                result = test_func(robot, scene_elements.get(scenario_name.replace("_", " "), []))
                results[scenario_name] = result
            except Exception as e:
                print(f"Error in {scenario_name}: {e}")
                results[scenario_name] = {"error": str(e)}

        # Print summary
        print("\n" + "="*50)
        print("NAVIGATION TEST SUMMARY")
        print("="*50)
        for scenario, result in results.items():
            if isinstance(result, dict) and "error" not in result:
                success = result.get("success", False)
                print(f"{scenario}: {'✓ PASS' if success else '✗ FAIL'}")
            else:
                print(f"{scenario}: ERROR - {result.get('error', 'Unknown error')}")

        return results
