# Collision Avoidance and Safety Constraints

## Introduction

Collision avoidance and safety constraints are critical for humanoid robot navigation, especially in human-populated environments. Unlike wheeled robots, humanoid robots must navigate with additional safety considerations due to their human-like scale, potential for causing harm to humans, and the risk of falling and damaging themselves. This chapter covers the implementation of robust collision avoidance systems and safety constraints for humanoid robot navigation.

## Humanoid-Specific Safety Considerations

### Physical Safety Factors
- **Human Proximity**: Humanoid robots operate at human eye level and must avoid collisions with humans
- **Fall Risk**: Humanoid robots can fall, potentially causing damage to themselves and surroundings
- **Manipulation Safety**: Arms and hands can cause harm if not properly constrained
- **Dynamic Balance**: Loss of balance can lead to unsafe situations

### Social Safety Factors
- **Personal Space**: Respecting human personal space and comfort zones
- **Social Navigation**: Following social norms for navigation in crowded areas
- **Predictable Behavior**: Ensuring robot behavior is predictable to humans
- **Emergency Protocols**: Safe stopping procedures when humans are at risk

### Safety Zones Classification
```python
# Example: Safety zone classification for humanoid robots
from enum import Enum
from dataclasses import dataclass
from typing import List, Tuple
import numpy as np

class SafetyZone(Enum):
    """Classification of safety zones around humans and obstacles"""
    DANGEROUS = 0      # Immediate collision risk (0.0-0.3m)
    CAUTION = 1        # Potential risk (0.3-0.8m)
    SOCIAL = 2         # Personal space (0.8-1.2m)
    COMFORTABLE = 3    # Social interaction zone (1.2-3.6m)

@dataclass
class SafetyRegion:
    """Represents a safety region around an object/human"""
    center: Tuple[float, float, float]  # x, y, z coordinates
    radius: float                       # radius of influence
    zone_type: SafetyZone               # safety classification
    priority: int                       # response priority (1=highest, 5=lowest)
    moving: bool                        # whether object is moving

class SafetyZoneClassifier:
    """Classifies safety zones for collision avoidance"""

    def __init__(self):
        # Define zone boundaries (in meters)
        self.zone_boundaries = {
            SafetyZone.DANGEROUS: 0.3,
            SafetyZone.CAUTION: 0.8,
            SafetyZone.SOCIAL: 1.2,
            SafetyZone.COMFORTABLE: 3.6
        }

        # Safety response priorities
        self.response_priorities = {
            SafetyZone.DANGEROUS: 1,   # Highest priority
            SafetyZone.CAUTION: 2,
            SafetyZone.SOCIAL: 3,
            SafetyZone.COMFORTABLE: 5  # Lowest priority
        }

    def classify_region(self, distance: float, object_type: str = "unknown") -> SafetyRegion:
        """Classify a region based on distance and object type"""
        # Determine zone based on distance
        zone_type = self.get_zone_type(distance)

        # Adjust for object type
        if object_type == "human":
            # Humans get more conservative safety margins
            if distance < self.zone_boundaries[SafetyZone.CAUTION]:
                zone_type = SafetyZone.DANGEROUS
            elif distance < self.zone_boundaries[SafetyZone.SOCIAL]:
                zone_type = SafetyZone.CAUTION
        elif object_type == "moving_object":
            # Moving objects get higher priority
            if zone_type == SafetyZone.COMFORTABLE:
                zone_type = SafetyZone.SOCIAL

        priority = self.response_priorities[zone_type]

        return SafetyRegion(
            center=(0, 0, 0),  # Placeholder
            radius=distance,
            zone_type=zone_type,
            priority=priority,
            moving=object_type == "moving_object"
        )

    def get_zone_type(self, distance: float) -> SafetyZone:
        """Get safety zone type based on distance"""
        if distance <= self.zone_boundaries[SafetyZone.DANGEROUS]:
            return SafetyZone.DANGEROUS
        elif distance <= self.zone_boundaries[SafetyZone.CAUTION]:
            return SafetyZone.CAUTION
        elif distance <= self.zone_boundaries[SafetyZone.SOCIAL]:
            return SafetyZone.SOCIAL
        else:
            return SafetyZone.COMFORTABLE

    def calculate_influence_cost(self, region: SafetyRegion) -> float:
        """Calculate cost for costmap based on safety region"""
        # Higher cost for higher priority (more dangerous) zones
        distance_factor = max(0.0, 1.0 - region.radius / 3.0)  # Decreases with distance
        priority_factor = 5.0 / region.priority  # Higher for higher priority

        # Moving objects get higher cost
        motion_factor = 1.5 if region.moving else 1.0

        cost = min(254.0, distance_factor * priority_factor * motion_factor * 100.0)
        return cost
```

### 3D Collision Detection
```python
# Example: 3D collision detection for humanoid robots
import numpy as np
from scipy.spatial.distance import cdist
from geometry_msgs.msg import Point
from sensor_msgs.msg import PointCloud2
import sensor_msgs_py.point_cloud2 as pc2

class ThreeDCollisionDetector:
    """3D collision detection for humanoid robots"""

    def __init__(self):
        # Robot dimensions and safety margins
        self.robot_height = 1.6  # meters (humanoid height)
        self.robot_radius = 0.3  # meters (cylindrical approximation)
        self.safety_margin = 0.2  # additional safety buffer

        # Collision detection parameters
        self.collision_threshold = 0.1  # meters
        self.warning_threshold = 0.5   # meters

    def detect_collisions_3d(self, pointcloud: PointCloud2, robot_pose) -> List[dict]:
        """Detect potential collisions in 3D space"""
        # Convert PointCloud2 to numpy array
        points = self.pointcloud_to_numpy(pointcloud)

        # Define robot bounding volume (cylinder)
        robot_volume = self.define_robot_volume(robot_pose)

        # Check for collisions
        collisions = []
        warnings = []

        for point in points:
            # Calculate distance to robot center (horizontal plane)
            horizontal_dist = np.sqrt((point[0] - robot_pose.position.x)**2 +
                                    (point[1] - robot_pose.position.y)**2)

            # Check horizontal collision (considering robot radius and safety margin)
            if horizontal_dist < (self.robot_radius + self.safety_margin):
                # Check vertical collision (height-wise)
                vertical_dist = abs(point[2] - robot_pose.position.z)

                # Humanoid robots are tall, so check multiple height levels
                if vertical_dist < self.robot_height:
                    # Potential collision detected
                    collision_info = {
                        'point': point,
                        'distance': np.sqrt(horizontal_dist**2 + vertical_dist**2),
                        'collision_type': self.classify_collision_type(point, robot_pose),
                        'severity': self.calculate_collision_severity(point, robot_pose),
                        'recommended_action': self.determine_recommended_action(point, robot_pose)
                    }

                    if collision_info['distance'] < self.collision_threshold:
                        collisions.append(collision_info)
                    elif collision_info['distance'] < self.warning_threshold:
                        warnings.append(collision_info)

        return collisions, warnings

    def define_robot_volume(self, robot_pose):
        """Define 3D volume occupied by robot"""
        # Approximate as cylinder with safety margin
        center = np.array([robot_pose.position.x, robot_pose.position.y, robot_pose.position.z])
        radius = self.robot_radius + self.safety_margin
        height = self.robot_height

        return {
            'center': center,
            'radius': radius,
            'height': height
        }

    def classify_collision_type(self, point, robot_pose):
        """Classify type of potential collision"""
        # Calculate relative position
        rel_x = point[0] - robot_pose.position.x
        rel_y = point[1] - robot_pose.position.y
        rel_z = point[2] - robot_pose.position.z

        # Determine collision location on robot
        if abs(rel_z) < 0.5:  # Lower body / legs
            return "lower_body_collision"
        elif 0.5 <= abs(rel_z) <= 1.2:  # Mid body / torso
            return "torso_collision"
        else:  # Head / upper body
            return "upper_body_collision"

    def calculate_collision_severity(self, point, robot_pose):
        """Calculate severity of potential collision"""
        # Distance-based severity (closer = more severe)
        dist = np.sqrt(
            (point[0] - robot_pose.position.x)**2 +
            (point[1] - robot_pose.position.y)**2 +
            (point[2] - robot_pose.position.z)**2
        )

        # Severity increases exponentially as distance decreases
        severity = max(0.0, 1.0 - dist / self.warning_threshold)
        return min(1.0, severity)  # Clamp to 0-1

    def determine_recommended_action(self, point, robot_pose):
        """Determine recommended action based on collision"""
        severity = self.calculate_collision_severity(point, robot_pose)

        if severity > 0.8:
            return "immediate_stop"
        elif severity > 0.5:
            return "emergency_avoidance"
        elif severity > 0.2:
            return "cautious_navigation"
        else:
            return "normal_navigation"

    def pointcloud_to_numpy(self, pointcloud_msg):
        """Convert PointCloud2 message to numpy array"""
        points_list = []
        for point in pc2.read_points(pointcloud_msg, field_names=("x", "y", "z"), skip_nans=True):
            points_list.append([point[0], point[1], point[2]])

        return np.array(points_list)
```

class SociallyAwareCollisionAvoidance:
    """Collision avoidance considering social norms and human comfort"""

    def __init__(self):
        # Social navigation parameters
        self.personal_space_radius = 0.8  # meters
        self.social_zone_radius = 1.2    # meters
        self.public_zone_radius = 3.6    # meters

        self.human_trackers = {}  # {track_id: tracker}
        self.social_zones = []    # List of active social zones

        """Update tracking of humans in environment"""

            track_id = human.id

            if track_id not in self.human_trackers:
                # Initialize new tracker
                self.human_trackers[track_id] = HumanTracker(human.position)

            # Update existing tracker
            self.human_trackers[track_id].update(human.position, human.velocity)

        # Remove old trackers that haven't been seen
        current_tracks = {h.id for h in humans_detected}
        expired_tracks = set(self.human_trackers.keys()) - current_tracks

        for track_id in expired_tracks:
            del self.human_trackers[track_id]

        """Create costmap overlay for social zones"""
        # Create costmap based on human positions and social norms
        social_costmap = np.zeros((100, 100))  # Example grid

            human_pos = tracker.get_current_position()

            # Add personal space cost (highest cost)
            self.add_personal_space_cost(social_costmap, human_pos)

            # Add social zone cost (medium cost)
            self.add_social_zone_cost(social_costmap, human_pos)

        return social_costmap

    def add_personal_space_cost(self, costmap, human_pos):
        """Add high cost for personal space violation"""
        # Convert world coordinates to grid coordinates
        grid_x, grid_y = self.world_to_grid(human_pos.x, human_pos.y)

        # Create circular cost area around human
        radius = int(self.personal_space_radius / self.costmap_resolution)

        for dx in range(-radius, radius + 1):
            for dy in range(-radius, radius + 1):
                if dx*dx + dy*dy <= radius*radius:
                    x_idx = grid_x + dx
                    y_idx = grid_y + dy

                    if 0 <= x_idx < costmap.shape[0] and 0 <= y_idx < costmap.shape[1]:
                        # High cost for personal space
                        costmap[x_idx, y_idx] = max(costmap[x_idx, y_idx], 200)

        # Similar to personal space but with lower cost and larger radius
        grid_x, grid_y = self.world_to_grid(human_pos.x, human_pos.y)
        radius = int(self.social_zone_radius / self.costmap_resolution)

        for dx in range(-radius, radius + 1):
            for dy in range(-radius, radius + 1):
                dist_sq = dx*dx + dy*dy
                if dist_sq <= radius*radius:
                    # Only apply cost outside personal space
                    personal_radius_sq = int(self.personal_space_radius / self.costmap_resolution)**2
                    if dist_sq > personal_radius_sq:
                        x_idx = grid_x + dx
                        y_idx = grid_y + dy

                        if 0 <= x_idx < costmap.shape[0] and 0 <= y_idx < costmap.shape[1]:
                            # Medium cost for social zone
                            costmap[x_idx, y_idx] = max(costmap[x_idx, y_idx], 100)

        """Plan path considering social navigation norms"""
        # Update human tracking
        self.update_human_tracking(humans_nearby)

        # Create social costmap overlay
        social_costmap = self.create_social_costmap_overlay()

        # Plan path using modified costmap
        path = self.plan_path_with_costmap_overlay(start, goal, social_costmap)

        # Post-process path for social navigation
        socially_aware_path = self.adapt_path_for_social_navigation(path, humans_nearby)

        return socially_aware_path

    def adapt_path_for_social_navigation(self, path, humans):
        """Adapt path to follow social navigation norms"""
        adapted_path = []

        for i, pose in enumerate(path):
            if i == 0:
                adapted_path.append(pose)
                continue

            prev_pose = adapted_path[-1]

            # Check if path passes too close to humans
            too_close_to_humans = False
            for human in humans:
                dist = self.calculate_distance_2d(pose.position, human.position)
                if dist < self.social_zone_radius:
                    too_close_to_humans = True
                    break

            if too_close_to_humans:
                # Adjust path to go around humans
                detour_pose = self.calculate_detour_around_human(pose, prev_pose, humans)
                adapted_path.append(detour_pose)
            else:
                adapted_path.append(pose)

        # Find the best detour direction to avoid humans
        best_detour = current_pose  # Default to original pose
        min_human_distance = float('inf')

        # Try different detour directions
        detour_offsets = [
            (0.3, 0.3),   # Northeast
            (0.3, -0.3),  # Southeast

            (-0.3, -0.3), # Southwest
            (0.5, 0),     # East
            (-0.5, 0),    # West
            (0, 0.5),     # North
            (0, -0.5),    # South

            detour_pose = Pose()
            detour_pose.position.x = current_pose.position.x + dx
            detour_pose.position.y = current_pose.position.y + dy
            detour_pose.position.z = current_pose.position.z

            # Calculate minimum distance to any human
            min_dist_to_humans = min(
                self.calculate_distance_2d(detour_pose.position, h.position) for h in humans

            if min_dist_to_humans > min_human_distance:

class HumanTracker:
    """Track individual human movement and predict future positions"""

    def __init__(self, initial_position):
        self.position_history = [initial_position]
        self.velocity = None
        self.predicted_path = []

    def update(self, position, velocity=None):
        """Update tracker with new position and velocity"""
        self.position_history.append(position)

        if velocity is not None:
            self.velocity = velocity
        else:
            # Calculate velocity from history if not provided
            if len(self.position_history) >= 2:
                dt = 0.1  # Assume 10Hz update rate
                prev_pos = self.position_history[-2]
                self.velocity = (
                    (position.x - prev_pos.x) / dt,
                    (position.y - prev_pos.y) / dt,
                    (position.z - prev_pos.z) / dt
                )

    def get_current_position(self):
        """Get current estimated position"""
        return self.position_history[-1] if self.position_history else None

    def predict_future_position(self, time_ahead):
        """Predict human position in the future"""
        if self.velocity is not None and len(self.position_history) >= 1:
            current_pos = self.position_history[-1]
            predicted_x = current_pos.x + self.velocity[0] * time_ahead
            predicted_y = current_pos.y + self.velocity[1] * time_ahead
            predicted_z = current_pos.z + self.velocity[2] * time_ahead

            return Point(predicted_x, predicted_y, predicted_z)

        return self.get_current_position()  # Fallback to current position

# Example: Emergency safety systems
class EmergencySafetySystem:
    """Emergency safety system for humanoid robots"""

    def __init__(self):
        self.emergency_level = 0  # 0 = normal, 1 = caution, 2 = warning, 3 = danger
        self.emergency_actions = []
        self.safety_engaged = False
        self.last_emergency_time = None
        self.emergency_cooldown = 2.0  # seconds between emergency actions

        # Safety parameters
        self.collision_imminent_threshold = 0.2  # meters
        self.high_risk_threshold = 0.5           # meters
        self.balance_threshold = 0.1             # stability measure

    def assess_emergency_situation(self, collision_data, balance_data, humans_nearby):
        """Assess current emergency level"""
        # Evaluate collision risk
        collision_risk = self.evaluate_collision_risk(collision_data)

        # Evaluate balance risk
        balance_risk = self.evaluate_balance_risk(balance_data)

        # Evaluate human safety risk
        human_risk = self.evaluate_human_safety_risk(humans_nearby)

        # Combine risks to determine overall emergency level
        overall_risk = max(collision_risk, balance_risk, human_risk)

        if overall_risk >= 0.9:
            return 3  # Danger
        elif overall_risk >= 0.7:
            return 2  # Warning
        elif overall_risk >= 0.4:
            return 1  # Caution
        else:
            return 0  # Normal

    def evaluate_collision_risk(self, collision_data):
        """Evaluate collision risk from collision detection data"""
        if not collision_data:
            return 0.0

        # Calculate risk based on closest collision threat
        min_distance = min([data['distance'] for data in collision_data], default=float('inf'))

        if min_distance <= self.collision_imminent_threshold:
            return 0.9  # Very high risk
        elif min_distance <= self.high_risk_threshold:
            return 0.7  # High risk
        else:
            # Lower risk based on distance
            risk_factor = (self.high_risk_threshold - min_distance) / self.high_risk_threshold
            return max(0.0, risk_factor)

    def evaluate_balance_risk(self, balance_data):
        """Evaluate risk based on robot balance state"""
        if not balance_data:
            return 0.0

        # Balance risk increases as stability decreases
        stability = balance_data.get('stability', 1.0)  # 0 = unstable, 1 = stable
        lean_angle = balance_data.get('lean_angle', 0.0)  # radians

        # Higher risk with higher lean angle
        lean_risk = min(1.0, lean_angle / (np.pi / 6))  # Risk at 30-degree lean

        # Combined risk
        balance_risk = max(1.0 - stability, lean_risk)
        return balance_risk

    def evaluate_human_safety_risk(self, humans_nearby):
        """Evaluate risk to humans in the vicinity"""
        if not humans_nearby:
            return 0.0

        max_human_risk = 0.0

        for human in humans_nearby:
            distance = self.calculate_human_distance(human)

            if distance < 0.3:  # Very close
                human_risk = 0.9
            elif distance < 0.8:  # Close
                human_risk = 0.7
            elif distance < 1.5:  # Within reach
                human_risk = 0.4
            else:  # Far away
                human_risk = 0.1

            max_human_risk = max(max_human_risk, human_risk)

        return max_human_risk

    def trigger_emergency_response(self, emergency_level, reason="Unknown"):
        """Trigger appropriate emergency response based on level"""
        current_time = time.time()

        # Check cooldown to prevent spam
        if (self.last_emergency_time and
            current_time - self.last_emergency_time < self.emergency_cooldown):
            return

        self.last_emergency_time = current_time

        if emergency_level >= 3:  # Danger
            self.execute_danger_protocol(reason)
        elif emergency_level >= 2:  # Warning
            self.execute_warning_protocol(reason)
        elif emergency_level >= 1:  # Caution
            self.execute_caution_protocol(reason)

    def execute_danger_protocol(self, reason):
        """Execute protocol for dangerous situations"""
        self.get_logger().error(f"DANGER: Emergency protocol activated - {reason}")

        # Immediate actions
        self.immediate_stop()           # Stop all motion
        self.activate_balance_control() # Engage balance control
        self.sound_alarm()              # Alert nearby humans

        angular_speed = np.sqrt(cmd_vel.angular.x**2 + cmd_vel.angular.y**2 + cmd_vel.angular.z**2)

        # Compare with safety limits
        max_safe_linear = 0.3  # m/s
        max_safe_angular = 0.5  # rad/s

        if linear_speed > max_safe_linear or angular_speed > max_safe_angular:
            self.get_logger().warn(f"Unsafe velocity command detected: linear={linear_speed}, angular={angular_speed}")
            return False

        return True
