# Global and Local Planners for Humanoid Robots

## Introduction

Path planning for humanoid robots requires specialized algorithms that account for the unique challenges of bipedal locomotion, balance constraints, and human-scale environments. Unlike wheeled robots, humanoid robots must consider gait patterns, balance maintenance, and the complex kinematics of walking, which significantly affect both global and local planning processes.

## Humanoid-Specific Planning Challenges

### Bipedal Locomotion Constraints
- **Gait Patterns**: Must follow stable walking patterns (single support, double support phases)
- **Balance Maintenance**: Path must allow for stable balance throughout execution
- **Step Constraints**: Discrete step planning with size and placement limitations
- **Dynamic Stability**: Center of Mass (CoM) must remain within stable regions

### Humanoid-Specific Constraints
- **Foot Placement**: Precise foot placement required for stability
- **Step Height**: Limited step height capabilities
- **Turning Radius**: Different turning mechanics compared to wheeled robots
- **Terrain Negotiation**: Stairs, slopes, and uneven terrain challenges

```python
from nav2_core.controller import Controller
from geometry_msgs.msg import Twist, PoseStamped
from nav_msgs.msg import Path

class GaitConstrainedLocalPlanner(Controller):
    def __init__(self):
        self.controller_frequency = 20.0  # Hz
        self.linear_vel_max = 0.3  # m/s (slower for stability)
        self.angular_vel_max = 0.5  # rad/s
        self.min_step_length = 0.15  # Minimum step length
        self.max_step_length = 0.6   # Maximum step length
        self.gait_frequency = 2.0    # Hz

    def setPlan(self, path: Path):
        """Set the path for the controller"""
        self.path = self.adapt_path_for_local_planning(path)
        self.current_waypoint_index = 0
        self.gait_phase = 0.0
        self.next_gait_time = self.get_clock().now()

    def computeVelocityCommands(self, pose: PoseStamped, velocity: Twist, goal_checker):
        """Compute velocity commands with gait constraints"""
        # Update gait phase
        self.update_gait_phase()

        # Check current balance state
        balance_state = self.get_current_balance_state()

        if not self.is_balanced(balance_state):
            # Emergency: recover balance immediately
            return self.balance_recovery_commands()

        # Get next waypoint to track
        next_waypoint = self.get_next_waypoint(pose)

        if next_waypoint is None:
            # Goal reached
            return Twist()  # Zero velocity

        # Compute commands with gait constraints
        cmd_vel = self.compute_gait_constrained_commands(
            pose, next_waypoint, velocity, balance_state
        )

        # Apply safety limits
        cmd_vel = self.apply_gait_constraints(cmd_vel, balance_state)

        return cmd_vel

    def compute_gait_constrained_commands(self, current_pose, target_waypoint, velocity, balance_state):
        """Compute velocity commands considering gait constraints"""
        # Calculate desired heading to target
        dx = target_waypoint.pose.position.x - current_pose.pose.position.x
        dy = target_waypoint.pose.position.y - current_pose.pose.position.y

        desired_angle = np.arctan2(dy, dx)
        current_angle = self.get_yaw_from_pose(current_pose.pose.orientation)

        # Calculate angular error
        angle_error = self.normalize_angle(desired_angle - current_angle)

        # Calculate distance to target
        distance = np.sqrt(dx*dx + dy*dy)

        # Compute commands
        cmd_vel = Twist()

        # Linear velocity: proportional to distance, limited by gait constraints
        cmd_vel.linear.x = min(
            self.linear_vel_max * 0.8,  # Conservative for stability
            max(0.05, distance * 1.0)   # Minimum speed to keep moving
        )

        # Angular velocity: proportional to angle error
        cmd_vel.angular.z = angle_error * 1.0

        # Apply gait-specific constraints
        cmd_vel = self.apply_gait_specific_constraints(cmd_vel, distance, angle_error)

        return cmd_vel

    def apply_gait_specific_constraints(self, cmd_vel, distance, angle_error):
        """Apply gait-specific constraints to velocity commands"""
        # Adjust commands based on gait phase
        gait_factor = self.get_gait_phase_factor()
        cmd_vel.linear.x *= gait_factor
        cmd_vel.angular.z *= gait_factor

        # Adjust for turning (may need different step patterns)
        if abs(angle_error) > 0.2:  # Significant turning required
            cmd_vel.linear.x *= 0.7  # Reduce speed during turns

        # Ensure commands are compatible with step constraints
        max_linear_for_step = self.max_step_length * self.gait_frequency
        cmd_vel.linear.x = min(cmd_vel.linear.x, max_linear_for_step)

        return cmd_vel

    def get_gait_phase_factor(self):
        """Get factor based on current gait phase"""
        # During single support phase, be more conservative
        # During double support phase, allow more aggressive motion
        phase = self.gait_phase
        if 0.0 <= phase < 0.5:  # Double support phase (more stable)
            return 1.0
        else:  # Single support phase (less stable)
            return 0.7

    def update_gait_phase(self):
        """Update gait phase based on time"""
        current_time = self.get_clock().now().nanoseconds / 1e9
        gait_period = 1.0 / self.gait_frequency
        self.gait_phase = (current_time % gait_period) / gait_period

    def apply_gait_constraints(self, cmd_vel, balance_state):
        """Apply final gait and balance constraints"""
        # Ensure commands don't exceed safe limits
        cmd_vel.linear.x = max(-self.linear_vel_max, min(self.linear_vel_max, cmd_vel.linear.x))
        cmd_vel.angular.z = max(-self.angular_vel_max, min(self.angular_vel_max, cmd_vel.angular.z))

        # Additional balance-based adjustments
        stability_factor = balance_state.get_stability_factor()
        cmd_vel.linear.x *= stability_factor
        cmd_vel.angular.z *= stability_factor

        return cmd_vel
```

```python
                    left_foot_pose, right_foot_pose, path_pose

                left_foot_pose = new_left_pose

                    right_foot_pose, left_foot_pose, path_pose
```

```python
    def interruptible_planning(self, start, goal, interrupt_flag):
        """Planning that can be interrupted for real-time requirements"""
        # Implement planning with periodic checks for interruption
        path = []
        current = start

        while not self.at_goal(current, goal) and not interrupt_flag.is_set():
            # Plan next segment
            next_segment = self.plan_next_segment(current, goal)

            if next_segment:
                path.extend(next_segment)
                current = next_segment[-1]
            else:
                # No valid path found
                break

            # Check for interruption
            if interrupt_flag.is_set():
                break

        return path
```

```python
# Example: Integration with Nav2 navigation stack
from nav2_behavior_tree.tree_node import TreeNode
from nav2_msgs.action import ComputePathToPose
from geometry_msgs.msg import PoseStamped
import py_trees

class ComputeGaitAwarePathToPose(TreeNode):
    def __init__(self, name, goal_service, planner_id='GaitAwarePlanner'):
        super().__init__(name)
        self._goal_service = goal_service
        self._planner_id = planner_id
        self._path = None

    def tick(self):
        """Compute path with humanoid-specific constraints"""
        if self._path is not None:
            return py_trees.common.Status.SUCCESS

        # Get goal from blackboard
        goal = self.attach_blackboard_client(name=f"{self.name}_blackboard")
        goal.register_key(key="goal", access=py_trees.common.Access.READ)

        # Create path request with humanoid constraints
        path_request = ComputePathToPose.Request()
        path_request.start = self.get_robot_position()
        path_request.goal = goal.goal
        path_request.planner_id = self._planner_id
        path_request.use_start = True

        # Add humanoid-specific parameters
        path_request.options = {
            'gait_frequency': 2.0,
            'balance_threshold': 0.8,
            'step_length_max': 0.6
        }

        try:
            # Call path planning service
            future = self._goal_service(path_request)
            if future.done():
                result = future.result()
                if result.error_code == 0:
                    self._path = result.path
                    return py_trees.common.Status.SUCCESS
                else:
                    self._feedback_message = f"Path planning failed: {result.error_code}"
                    return py_trees.common.Status.FAILURE
            else:
                return py_trees.common.Status.RUNNING

        except Exception as e:
            self._feedback_message = f"Path planning service call failed: {e}"
            return py_trees.common.Status.FAILURE
```
