# Navigation Performance and Safety Validation

## Introduction

Navigation performance and safety validation is critical for humanoid robots operating in human environments. Unlike wheeled robots, humanoid robots must navigate while maintaining balance, which adds complexity to both performance metrics and safety considerations. This chapter outlines comprehensive validation approaches for Isaac-based navigation systems in humanoid robots.

## Performance Metrics for Humanoid Navigation

### Standard Navigation Metrics
- **Success Rate**: Percentage of navigation goals achieved
- **Time to Goal**: Duration from start to successful navigation
- **Path Length**: Actual distance traveled vs. optimal path
- **Efficiency Ratio**: Optimal path length / actual path length
- **Average Speed**: Effective navigation speed during missions

### Humanoid-Specific Performance Metrics
- **Balance Maintenance**: Time spent in stable vs. unstable states
- **Gait Consistency**: Regularity of step patterns during navigation
- **Energy Efficiency**: Power consumption relative to distance traveled
- **Social Compliance**: Adherence to social navigation norms
- **Human Comfort**: Subjective measures of how comfortable humans feel around the robot

### Real-Time Performance Indicators
- **Processing Latency**: Delay between sensor input and navigation command
- **Control Frequency**: Actual vs. target control loop frequency
- **Memory Usage**: GPU and CPU memory consumption during navigation
- **Thermal Management**: Component temperatures during sustained operation

### Safety Zones Classification
```python
# Example: Safety zone classification for humanoid navigation
from enum import Enum
from dataclasses import dataclass
from typing import List, Tuple

class SafetyZone(Enum):
    """Classification of safety zones around humans and obstacles"""
    DANGEROUS = 0      # Immediate collision risk (0.0-0.3m)
    CAUTION = 1        # Potential risk (0.3-0.8m)
    SOCIAL = 2         # Personal space (0.8-1.2m)
    COMFORTABLE = 3    # Social interaction zone (1.2-3.6m)

@dataclass
class SafetyRegion:
    """Represents a safety region around an object/human"""
    center: Tuple[float, float, float]  # x, y, z coordinates
    radius: float                       # radius of influence
    zone_type: SafetyZone               # safety classification
    priority: int                       # response priority (1=highest, 5=lowest)
    moving: bool                        # whether object is moving

class SafetyZoneClassifier:
    """Classifies safety zones for navigation safety"""

    def __init__(self):
        # Define zone boundaries (in meters)
        self.zone_boundaries = {
            SafetyZone.DANGEROUS: 0.3,
            SafetyZone.CAUTION: 0.8,
            SafetyZone.SOCIAL: 1.2,
            SafetyZone.COMFORTABLE: 3.6
        }

        # Safety response priorities
        self.response_priorities = {
            SafetyZone.DANGEROUS: 1,   # Highest priority
            SafetyZone.CAUTION: 2,
            SafetyZone.SOCIAL: 3,
            SafetyZone.COMFORTABLE: 5  # Lowest priority
        }

    def classify_region(self, distance: float, object_type: str = "unknown") -> SafetyRegion:
        """Classify a region based on distance and object type"""
        # Determine zone based on distance
        zone_type = self.get_zone_type(distance)

        # Adjust for object type
        if object_type == "human":
            # Humans get more conservative safety margins
            if distance < self.zone_boundaries[SafetyZone.CAUTION]:
                zone_type = SafetyZone.DANGEROUS
            elif distance < self.zone_boundaries[SafetyZone.SOCIAL]:
                zone_type = SafetyZone.CAUTION
        elif object_type == "moving_object":
            # Moving objects get higher priority
            if zone_type == SafetyZone.COMFORTABLE:
                zone_type = SafetyZone.SOCIAL

        priority = self.response_priorities[zone_type]

        return SafetyRegion(
            center=(0, 0, 0),  # Placeholder
            radius=distance,
            zone_type=zone_type,
            priority=priority,
            moving=object_type == "moving_object"
        )

    def get_zone_type(self, distance: float) -> SafetyZone:
        """Get safety zone type based on distance"""
        if distance <= self.zone_boundaries[SafetyZone.DANGEROUS]:
            return SafetyZone.DANGEROUS
        elif distance <= self.zone_boundaries[SafetyZone.CAUTION]:
            return SafetyZone.CAUTION
        elif distance <= self.zone_boundaries[SafetyZone.SOCIAL]:
            return SafetyZone.SOCIAL
        else:
            return SafetyZone.COMFORTABLE

    def calculate_influence_cost(self, region: SafetyRegion) -> float:
        """Calculate cost for costmap based on safety region"""
        # Higher cost for higher priority (more dangerous) zones
        distance_factor = max(0.0, 1.0 - region.radius / 3.0)  # Decreases with distance
        priority_factor = 5.0 / region.priority  # Higher for higher priority

        # Moving objects get higher cost
        motion_factor = 1.5 if region.moving else 1.0

        cost = min(254.0, distance_factor * priority_factor * motion_factor * 100.0)
        return cost
```

        robot.navigate_to_pose(path_to_goal)

        collision_count = 0

            current_pos = robot.get_position()

            min_dist = min([self.calculate_distance(current_pos, obs_pos) for obs_pos in obstacle_positions])
            min_distances.append(min_dist)

            # Check for collisions
            if min_dist < robot.radius:
                collision_count += 1

            time.sleep(0.1)  # Monitor at 10Hz

        execution_time = time.time() - start_time
        avg_min_distance = sum(min_distances) / len(min_distances) if min_distances else float('inf')

            "success": robot.navigation_successful(),
            "execution_time": execution_time,
            "collision_count": collision_count,
            "avg_min_distance": avg_min_distance,
            "closest_approach": closest_approach,
            "obstacle_count": len(obstacle_positions),
            "path_deviation": self.calculate_path_deviation(initial_path, robot.get_executed_path())
        }

        return results

                        "time": violation_time,
                        "distance": dist_to_human,
                        "human_id": human.id
                    })

            balance_metrics.append(current_balance)

            # Record path efficiency
            optimal_dist = self.calculate_distance(current_pos, goal_pose.position)

        execution_time = time.time() - start_time

        violation_count = len(social_violations)
        avg_balance = sum(balance_metrics) / len(balance_metrics) if balance_metrics else 0
        avg_path_efficiency = sum(path_efficiency) / len(path_efficiency) if path_efficiency else float('inf')

        results = {
            "success": robot.navigation_successful(),
            "execution_time": execution_time,
            "social_violations": violation_count,
            "avg_social_distance": np.mean([v['distance'] for v in social_violations]) if social_violations else float('inf'),
            "avg_balance_stability": avg_balance,
            "avg_path_efficiency": avg_path_efficiency,
            "human_agents_count": len(human_agents)
        }

        # Stop human agents
        for human in human_agents:

        """Validate navigation with moving obstacles"""
        # Start moving obstacles
        for obstacle in moving_obstacles:
            obstacle.start_movement()

        robot.navigate_to_pose(goal_pose)

        collision_count = 0
        path_inefficiency_count = 0

            # Check for collisions with moving obstacles
            for obstacle in moving_obstacles:

                dist = self.calculate_distance(current_pos, obs_pos)

                if dist < robot.radius:
                    collision_count += 1

            # Measure response time to obstacle appearance
            if robot.obstacle_detected_recently():
                response_start = time.time()
                while robot.obstacle_response_in_progress():
                    time.sleep(0.01)
                response_time = time.time() - response_start
                response_timeliness.append(response_time)

            time.sleep(0.1)

        # Stop moving obstacles
        for obstacle in moving_obstacles:
            obstacle.stop_movement()

        results = {
            "success": robot.navigation_successful(),
            "collision_count": collision_count,
            "path_inefficiency_events": path_inefficiency_count,
            "avg_response_time": np.mean(response_timeliness) if response_timeliness else float('inf'),
            "max_response_time": max(response_timeliness) if response_timeliness else float('inf'),
            "moving_obstacle_count": len(moving_obstacles)
        }

        if not initial_path or not executed_path:
            return float('inf')

        # Calculate how much the robot deviated from the initial plan
        # This would compare the actual path taken to the initially planned path
        deviation = 0.0
        # Implementation would calculate path similarity metrics

    def calculate_distance(self, pos1, pos2):
        """Calculate Euclidean distance between two positions"""
        dx = pos1[0] - pos2[0]

# Example: Safety validation tests
class SafetyValidator:
    def __init__(self):
        self.emergency_stop_triggered = False
        self.balance_recovery_count = 0
        self.collision_count = 0
        self.safety_violation_log = []

    def validate_collision_avoidance(self, robot, obstacle_scenarios):
        """Validate collision avoidance capabilities"""
        results = []

        for scenario in obstacle_scenarios:
            # Set up scenario
            self.setup_obstacle_scenario(robot, scenario)

            # Attempt navigation
            robot.navigate_to_goal(scenario.goal)

            # Monitor for safety events
            start_time = time.time()
            safety_events = []

            while not robot.navigation_complete() and (time.time() - start_time < 60):
                # Check for collisions
                if robot.check_for_collision():
                    self.collision_count += 1
                    safety_events.append({
                        "type": "collision",
                        "time": time.time() - start_time,
                        "position": robot.get_position()
                    })

                # Check for emergency stops
                if robot.emergency_stop_activated():
                    self.emergency_stop_triggered = True
                    safety_events.append({
                        "type": "emergency_stop",
                        "time": time.time() - start_time,
                        "position": robot.get_position()
                    })

                # Check for balance recovery
                if robot.balance_recovery_activated():
                    self.balance_recovery_count += 1
                    safety_events.append({
                        "type": "balance_recovery",
                        "time": time.time() - start_time,
                        "position": robot.get_position()
                    })

                time.sleep(0.1)

            scenario_result = {
                "scenario": scenario.name,
                "completed": robot.navigation_completed(),
                "successful": robot.navigation_successful(),
                "execution_time": time.time() - start_time,
                "safety_events": safety_events,
                "final_position": robot.get_final_position(),
                "goal_achieved": self.calculate_distance(
                    robot.get_final_position(),
                    scenario.goal
                ) < scenario.goal_tolerance
            }

            results.append(scenario_result)

        return results

    def validate_balance_recovery(self, robot, balance_disturbance_scenarios):
        """Validate balance recovery capabilities"""
        results = []

        for scenario in balance_disturbance_scenarios:
            # Apply disturbance
            robot.apply_external_disturbance(scenario.disturbance_force)

            # Monitor balance recovery
            start_time = time.time()
            recovery_started = False
            recovery_completed = False

            while (time.time() - start_time < 10.0):  # 10 second timeout
                current_balance = robot.get_balance_state()

                if not recovery_started and current_balance.stability < 0.5:
                    recovery_started = True
                    recovery_start_time = time.time()

                if (recovery_started and
                    current_balance.stability > robot.balance_threshold and
                    time.time() - recovery_start_time > 0.5):  # Stable for 0.5s
                    recovery_completed = True
                    break

                time.sleep(0.01)  # 100 Hz monitoring

            scenario_result = {
                "scenario": scenario.name,
                "disturbance_force": scenario.disturbance_force,
                "recovery_initiated": recovery_started,
                "recovery_completed": recovery_completed,
                "recovery_time": (time.time() - recovery_start_time) if recovery_completed else float('inf'),
                "balance_threshold": robot.balance_threshold
            }

            results.append(scenario_result)

        return results

    def validate_emergency_procedures(self, robot, emergency_scenarios):
        """Validate emergency stop and recovery procedures"""
        results = []

        for scenario in emergency_scenarios:
            # Set up emergency scenario
            if scenario.type == "immediate_stop":
                # Trigger immediate stop
                robot.emergency_stop()

                # Monitor stop performance
                start_time = time.time()
                stopped = False
                while time.time() - start_time < 2.0:  # 2 sec timeout for stop
                    if robot.get_velocity_magnitude() < 0.01:  # Nearly stopped
                        stopped = True
                        stop_time = time.time() - start_time
                        break
                    time.sleep(0.01)

                scenario_result = {
                    "type": "immediate_stop",
                    "stopped_successfully": stopped,
                    "stop_time": stop_time if stopped else 2.0,
                    "remaining_velocity": robot.get_velocity_magnitude() if not stopped else 0.0
                }

            elif scenario.type == "balance_recovery":
                # Force robot into unstable state
                robot.force_unstable_state()

                # Activate balance recovery
                robot.activate_balance_recovery()

                # Monitor recovery
                start_time = time.time()
                recovered = False
                while time.time() - start_time < 5.0:  # 5 sec timeout for recovery
                    if robot.is_balanced():
                        recovered = True
                        recovery_time = time.time() - start_time
                        break
                    time.sleep(0.01)

                scenario_result = {
                    "type": "balance_recovery",
                    "recovered_successfully": recovered,
                    "recovery_time": recovery_time if recovered else 5.0,
                    "final_balance_state": robot.get_balance_state()
                }

            results.append(scenario_result)

        return results

    def setup_obstacle_scenario(self, robot, scenario):
        """Set up obstacle scenario for testing"""
        # Add obstacles to robot's costmap
        for obstacle in scenario.obstacles:
            robot.add_static_obstacle(obstacle)

        # If dynamic obstacles, set up motion patterns
        for obstacle in scenario.dynamic_obstacles:
            robot.add_dynamic_obstacle(obstacle)

    def log_safety_violation(self, violation_type, details):
        """Log safety violation for analysis"""
        violation = {
            "timestamp": time.time(),
            "type": violation_type,
            "details": details,
            "robot_state": self.get_robot_state_snapshot()
        }
        self.safety_violation_log.append(violation)

    def get_robot_state_snapshot(self):
        """Get current robot state for violation logging"""
        return {
            "position": self.robot.get_position(),
            "velocity": self.robot.get_velocity(),
            "balance_state": self.robot.get_balance_state(),
            "navigation_state": self.robot.get_navigation_state(),
            "sensor_data": self.robot.get_recent_sensor_data()
        }

    def generate_safety_report(self):
        """Generate comprehensive safety validation report"""
        report = {
            "validation_date": time.strftime("%Y-%m-%d %H:%M:%S"),
            "total_collision_count": self.collision_count,
            "emergency_stop_triggers": self.emergency_stop_triggered,
            "balance_recovery_count": self.balance_recovery_count,
            "safety_violations": len(self.safety_violation_log),
            "violation_details": self.safety_violation_log,
            "recommendations": self.generate_safety_recommendations()
        }

        return report

    def generate_safety_recommendations(self):
        """Generate safety recommendations based on validation results"""
        recommendations = []

        if self.collision_count > 0:
            recommendations.append(
                f"Reduce collision count from {self.collision_count} to 0 through improved obstacle detection and path planning"
            )

        if self.balance_recovery_count > 5:  # Arbitrary threshold
            recommendations.append(
                f"Reduce balance recovery events from {self.balance_recovery_count} to improve stability"
            )

        if self.emergency_stop_triggered:
            recommendations.append(
                "Investigate causes of emergency stops and improve predictive avoidance"
            )

        return recommendations

    def validate_safety_state(self):
        """Main safety validation function"""
        with self.safety_lock:
            # Check all safety aspects
            collision_safe = self.check_collision_safety()
            human_safe = self.check_human_safety()
            balance_safe = self.check_balance_safety()
            command_safe = self.check_command_safety()

            # Overall safety status
            overall_safe = all([collision_safe, human_safe, balance_safe, command_safe])

            # Update safety status
            self.is_safe = overall_safe

            # Publish safety status
            safety_msg = Bool()
            safety_msg.data = overall_safe
            self.safety_status_pub.publish(safety_msg)

            # If unsafe, trigger appropriate responses
            if not overall_safe:
                self.handle_unsafe_condition(
                    collision_safe, human_safe, balance_safe, command_safe
                )

            # Publish balance status
            if self.current_imu_data:
                balance_stability = self.calculate_balance_stability(self.current_imu_data)
                balance_msg = Float32()
                balance_msg.data = balance_stability
                self.balance_status_pub.publish(balance_msg)

            # Create safety visualization
            self.create_safety_visualization()

    def check_collision_safety(self):
        """Check for collision safety based on obstacle detection"""
        if not self.current_pose or not self.detected_obstacles:
            return True  # Can't assess safety without data

        robot_pos = np.array([
            self.current_pose.pose.position.x,
            self.current_pose.pose.position.y,
            self.current_pose.pose.position.z
        ])

        for obstacle in self.detected_obstacles:
            obstacle_pos = np.array([
                obstacle.position.x,
                obstacle.position.y,
                obstacle.position.z
            ])

            distance = np.linalg.norm(robot_pos - obstacle_pos)

            if distance < self.collision_threshold:
                self.get_logger().warn(
                    f"Collision imminent! Distance: {distance:.2f}m < threshold: {self.collision_threshold}m"
                )
                return False

        return True

    def check_human_safety(self):
        """Check for human safety based on detected humans"""
        if not self.current_pose or not self.detected_humans:
            return True  # Can't assess safety without data

        robot_pos = np.array([
            self.current_pose.pose.position.x,
            self.current_pose.pose.position.y,
            self.current_pose.pose.position.z
        ])

        for human in self.detected_humans:
            human_pos = np.array([
                human.position.x,
                human.position.y,
                human.position.z
            ])

            distance = np.linalg.norm(robot_pos - human_pos)

            # Check personal space violation
            if distance < self.personal_space_radius:
                self.get_logger().warn(
                    f"Personal space violation! Distance: {distance:.2f}m < radius: {self.personal_space_radius}m"
                )
                return False

            # Check social zone proximity (warning level)
            if distance < self.social_zone_radius:
                self.get_logger().info(
                    f"In social zone. Distance: {distance:.2f}m, radius: {self.social_zone_radius}m"
                )

        return True

    def check_balance_safety(self):
        """Check for balance safety using IMU data"""
        if not self.current_imu_data:
            return True  # Can't assess without IMU data

        # Calculate tilt angles from IMU
        roll, pitch, _ = self.quaternion_to_euler(
            self.current_imu_data.orientation.x,
            self.current_imu_data.orientation.y,
            self.current_imu_data.orientation.z,
            self.current_imu_data.orientation.w
        )

        # Calculate tilt magnitude
        tilt_magnitude = math.sqrt(roll*roll + pitch*pitch)

        # Check if tilt exceeds safe threshold
        if tilt_magnitude > (math.pi / 4):  # 45 degrees
            self.get_logger().warn(
                f"Balance threshold exceeded! Tilt: {math.degrees(tilt_magnitude):.1f}°"
            )
            return False

        return True

    def check_command_safety(self):
        """Check if navigation commands are safe"""
        if not self.current_nav_cmd:
            return True  # No command to check

        # Check if commanded velocity is too high for current conditions
        linear_speed = math.sqrt(
            self.current_nav_cmd.linear.x**2 +

            self.current_nav_cmd.linear.z**2

        angular_speed = math.sqrt(
            self.current_nav_cmd.angular.x**2 +

            self.current_nav_cmd.angular.z**2
        )

        # Define safe speed limits based on environment and balance state
        max_linear_safe = self.calculate_max_safe_linear_speed()
        max_angular_safe = self.calculate_max_safe_angular_speed()

        if linear_speed > max_linear_safe or angular_speed > max_angular_safe:
            self.get_logger().warn(
                f"Unsafe command speeds! Linear: {linear_speed:.2f}, Angular: {angular_speed:.2f}"
            )
            return False

        return True

    def calculate_max_safe_linear_speed(self):
        """Calculate max safe linear speed based on current conditions"""
        # Reduce speed near humans or obstacles
        min_human_dist = float('inf')
        if self.detected_humans and self.current_pose:
            robot_pos = np.array([
                self.current_pose.pose.position.x,
                self.current_pose.pose.position.y
            ])
            for human in self.detected_humans:
                human_pos = np.array([human.position.x, human.position.y])
                dist = np.linalg.norm(robot_pos - human_pos)
                min_human_dist = min(min_human_dist, dist)

        if min_human_dist < 1.0:  # Near humans
            return 0.2  # Reduce speed
        elif min_human_dist < 2.0:
            return 0.4  # Moderate speed
        else:

    def calculate_max_safe_angular_speed(self):

        # For humanoid robots, reduce angular speed when turning to maintain balance
        if self.current_imu_data:

                self.current_imu_data.orientation.z,

            if tilt_magnitude > math.radians(10):  # 10 degree tilt
                return 0.3  # Reduce turning speed
            else:
                return 0.5  # Normal turning speed
        else:
            return 0.5

        max_acceptable_tilt = math.radians(15)  # 15 degrees max acceptable tilt
        stability = max(0.0, 1.0 - (tilt_magnitude / max_acceptable_tilt))

    def quaternion_to_euler(self, x, y, z, w):
        """Convert quaternion to Euler angles"""
        # Roll (x-axis rotation)
        sinr_cosp = 2 * (w * x + y * z)
        cosr_cosp = 1 - 2 * (x * x + y * y)
        roll = math.atan2(sinr_cosp, cosr_cosp)

        # Pitch (y-axis rotation)
        sinp = 2 * (w * y - z * x)
        if abs(sinp) >= 1:
            pitch = math.copysign(math.pi / 2, sinp)
        else:
            pitch = math.asin(sinp)

        # Yaw (z-axis rotation)
        siny_cosp = 2 * (w * z + x * y)
        cosy_cosp = 1 - 2 * (y * y + z * z)
        yaw = math.atan2(siny_cosp, cosy_cosp)

        return roll, pitch, yaw

    def calculate_orientation_error(self, estimated_pose, ground_truth_pose):
        """Calculate orientation error between estimate and ground truth"""

        est_rot = self.quaternion_to_rotation_matrix(estimated_pose.orientation)
        gt_rot = self.quaternion_to_rotation_matrix(ground_truth_pose.orientation)

        # Calculate rotation error using axis-angle representation
        rotation_matrix_error = est_rot.T @ gt_rot
        trace = np.trace(rotation_matrix_error)
        rotation_error = math.acos(max(-1, min(1, (trace - 1) / 2)))  # Clamp for numerical errors

        return rotation_error
