# VSLAM and Nav2 Integration

## Introduction

The integration between Visual SLAM (VSLAM) and Navigation2 (Nav2) is crucial for enabling autonomous humanoid robots to navigate effectively in unknown or dynamic environments. This integration combines the spatial awareness provided by VSLAM with the navigation capabilities of Nav2, creating a robust system for localization and path planning in human-centric environments.

## Architecture Overview

### Integration Architecture
```
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────┐

```

### Key Integration Points
- **Pose Estimation**: VSLAM provides localization for Nav2
- **Map Building**: VSLAM contributes to costmap generation
- **Loop Closure**: VSLAM enhances Nav2's global localization
- **Sensor Fusion**: Integration of visual and other sensors

class MultiSensorFusion:
    def __init__(self):
        # Sensor data sources
        self.vslam_data = None
        self.imu_data = None
        self.odom_data = None
        self.lidar_data = None

        # Fusion parameters
        self.vslam_confidence = 0.8
        self.imu_confidence = 0.7
        self.odom_confidence = 0.6
        self.lidar_confidence = 0.9

        # Covariance matrices for each sensor
        self.vslam_covariance = np.eye(6) * 0.1
        self.imu_covariance = np.eye(6) * 0.05
        self.odom_covariance = np.eye(6) * 0.2
        self.lidar_covariance = np.eye(6) * 0.08

    def fuse_sensor_data(self, vslam_pose, imu_data, odom_data, lidar_data):
        """Fuse data from multiple sensors using Kalman Filter approach"""
        # Convert sensor data to state vectors
        vslam_state = self.pose_to_state(vslam_pose)
        imu_state = self.imu_to_state(imu_data)
        odom_state = self.odom_to_state(odom_data)
        lidar_state = self.lidar_to_state(lidar_data)

        # Perform weighted fusion based on sensor reliability
        fused_state = self.weighted_state_fusion([
            (vslam_state, self.vslam_covariance),
            (imu_state, self.imu_covariance),
            (odom_state, self.odom_covariance),
            (lidar_state, self.lidar_covariance)
        ])

        return self.state_to_pose(fused_state)

    def weighted_state_fusion(self, sensor_states_and_covariances):
        """Weighted fusion of multiple sensor states"""
        # Use inverse covariance weighting (Kalman Filter approach)
        total_weighted_state = np.zeros(6)
        total_information_matrix = np.zeros((6, 6))

        for state, covariance in sensor_states_and_covariances:
            # Information matrix is inverse of covariance
            information = np.linalg.inv(covariance)
            weighted_state = information @ state

            total_weighted_state += weighted_state
            total_information_matrix += information

        # Compute fused state
        fused_covariance = np.linalg.inv(total_information_matrix)
        fused_state = fused_covariance @ total_weighted_state

        return fused_state

    def pose_to_state(self, pose):
        """Convert PoseStamped to state vector [x, y, z, roll, pitch, yaw]"""
        state = np.zeros(6)
        state[0] = pose.pose.position.x
        state[1] = pose.pose.position.y
        state[2] = pose.pose.position.z

        # Convert quaternion to euler angles
        quat = pose.pose.orientation
        roll, pitch, yaw = self.quaternion_to_euler(quat.x, quat.y, quat.z, quat.w)
        state[3] = roll
        state[4] = pitch
        state[5] = yaw

        return state

    def quaternion_to_euler(self, x, y, z, w):
        """Convert quaternion to euler angles"""
        # Roll (x-axis rotation)
        sinr_cosp = 2 * (w * x + y * z)
        cosr_cosp = 1 - 2 * (x * x + y * y)
        roll = np.arctan2(sinr_cosp, cosr_cosp)

        # Pitch (y-axis rotation)
        sinp = 2 * (w * y - z * x)
        if np.abs(sinp) >= 1:
            pitch = np.copysign(np.pi / 2, sinp)
        else:
            pitch = np.arcsin(sinp)

        # Yaw (z-axis rotation)
        siny_cosp = 2 * (w * z + x * y)
        cosy_cosp = 1 - 2 * (y * y + z * z)
        yaw = np.arctan2(siny_cosp, cosy_cosp)

        return roll, pitch, yaw

    def imu_to_state(self, imu_data):
        """Convert IMU data to state vector"""
        # Extract position from integrated IMU data
        # (simplified - in practice, this would involve integration)
        state = np.zeros(6)
        # Position from integration of acceleration
        # Orientation from IMU
        state[3] = np.arctan2(2*(imu_data.orientation.w*imu_data.orientation.x +
                                imu_data.orientation.y*imu_data.orientation.z),
                              1 - 2*(imu_data.orientation.x**2 +
                                    imu_data.orientation.y**2))
        state[4] = np.arcsin(2*(imu_data.orientation.w*imu_data.orientation.y -
                               imu_data.orientation.z*imu_data.orientation.x))
        state[5] = np.arctan2(2*(imu_data.orientation.w*imu_data.orientation.z +
                                imu_data.orientation.x*imu_data.orientation.y),
                              1 - 2*(imu_data.orientation.y**2 +
                                    imu_data.orientation.z**2))
