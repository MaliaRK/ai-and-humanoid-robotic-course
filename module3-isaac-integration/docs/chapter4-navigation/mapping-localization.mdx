# Mapping and Localization for Humanoid Robots

## Introduction

Mapping and localization are fundamental capabilities for autonomous humanoid robots, enabling them to understand their environment and navigate effectively. For humanoid robots, these capabilities must account for the unique challenges of bipedal locomotion, human-scale environments, and the need for real-time performance while maintaining balance.

## Mapping Challenges for Humanoid Robots

### Human-Scale Considerations
- **Height Advantage**: Humanoid robots can perceive environments from eye-level
- **Vertical Structure**: Ability to navigate stairs, ramps, and multi-level structures
- **Social Navigation**: Mapping for human-centric spaces and social norms
- **Personal Space**: Awareness of human personal space requirements

### Dynamic Environment Mapping
- **Moving Humans**: Tracking and predicting human movement patterns
- **Temporary Obstacles**: Furniture rearrangement, doors opening/closing
- **Lighting Changes**: Adapting to day/night and changing lighting conditions
- **Clutter Dynamics**: Objects moved by humans during robot operation

### Humanoid-Specific Localization Pipeline
```python
# Example: Humanoid localization pipeline
import numpy as np
from scipy.spatial.transform import Rotation as R
from geometry_msgs.msg import PoseWithCovarianceStamped
from sensor_msgs.msg import Imu, JointState
from tf2_ros import TransformListener
import particle_filters as pf  # Hypothetical particle filter library

class HumanoidLocalization:
    def __init__(self, node):
        self.node = node
        self.particle_filter = pf.ParticleFilter(state_dim=6)  # x, y, z, roll, pitch, yaw
        self.initialized = False
        self.pose_estimate = np.zeros(6)  # [x, y, z, roll, pitch, yaw]
        self.pose_covariance = np.eye(6) * 0.1  # Initial uncertainty

        # Motion model for humanoid gait
        self.motion_model = HumanoidMotionModel()

        # Sensor models
        self.lidar_model = LidarLikelihoodFieldModel()
        self.visual_model = VisualFeatureBasedModel()
        self.imu_model = IMUModel()
        self.kinematic_model = KinematicModel()

    def initialize_pose(self, initial_pose, uncertainty=0.1):
        """Initialize localization with initial pose estimate"""
        self.pose_estimate[:3] = [initial_pose.position.x,
                                  initial_pose.position.y,
                                  initial_pose.position.z]

        # Convert quaternion to euler angles
        quat = [initial_pose.orientation.x, initial_pose.orientation.y,
                initial_pose.orientation.z, initial_pose.orientation.w]
        r = R.from_quat(quat)
        self.pose_estimate[3:] = r.as_euler('xyz')

        # Initialize particle filter
        self.particle_filter.initialize(
            mean=self.pose_estimate,
            cov=np.eye(6) * uncertainty
        )
        self.initialized = True

    def update_localization(self, sensor_data, dt):
        """Update localization with new sensor data"""
        if not self.initialized:
            self.node.get_logger().warn("Localization not initialized")
            return

        # Predict step: motion model
        predicted_particles = self.predict_motion(dt, sensor_data)

        # Update step: sensor likelihood
        updated_weights = self.update_with_sensors(predicted_particles, sensor_data)

        # Resample particles
        resampled_particles = self.resample_particles(predicted_particles, updated_weights)

        # Estimate pose from particles
        self.pose_estimate, self.pose_covariance = self.estimate_pose_from_particles(resampled_particles)

        return self.pose_estimate, self.pose_covariance

    def predict_motion(self, dt, sensor_data):
        """Predict particle poses based on motion model"""
        # Get motion information from various sources
        imu_motion = self.extract_imu_motion(sensor_data.get('imu'))
        kinematic_motion = self.extract_kinematic_motion(sensor_data.get('joints'))
        visual_odometry = sensor_data.get('visual_odometry', np.zeros(6))

        # Combine motion estimates
        combined_motion = self.fuse_motion_estimates(
            imu_motion, kinematic_motion, visual_odometry
        )

        # Apply motion to particles
        predicted_particles = self.particle_filter.predict(
            motion=combined_motion,
            dt=dt,
            motion_noise=self.get_motion_noise()
        )

        return predicted_particles

    def extract_imu_motion(self, imu_data):
        """Extract motion information from IMU data"""
        if imu_data is None:
            return np.zeros(6)

        # Integrate IMU data for position/velocity changes
        linear_acc = np.array([
            imu_data.linear_acceleration.x,
            imu_data.linear_acceleration.y,
            imu_data.linear_acceleration.z
        ])

        angular_vel = np.array([
            imu_data.angular_velocity.x,
            imu_data.angular_velocity.y,
            imu_data.angular_velocity.z
        ])

        # Integrate to get pose change
        position_change = 0.5 * linear_acc * dt**2
        orientation_change = angular_vel * dt

        motion_vector = np.concatenate([position_change, orientation_change])
        return motion_vector

    def extract_kinematic_motion(self, joint_data):
        """Extract motion from leg kinematics"""
        if joint_data is None:
            return np.zeros(6)

        # Calculate body motion based on leg movements
        # This would involve forward kinematics and gait analysis
        body_motion = self.kinematic_model.calculate_body_motion(joint_data)
        return body_motion

    def update_with_sensors(self, particles, sensor_data):
        """Update particle weights based on sensor likelihood"""
        weights = np.ones(len(particles))

        # Update with LiDAR data
        if 'lidar' in sensor_data:
            lidar_weights = self.lidar_model.calculate_likelihood(
                particles, sensor_data['lidar']
            )
            weights *= lidar_weights

        # Update with visual features
        if 'features' in sensor_data:
            visual_weights = self.visual_model.calculate_likelihood(
                particles, sensor_data['features']
            )
            weights *= visual_weights

        # Update with IMU data
        if 'imu' in sensor_data:
            imu_weights = self.imu_model.calculate_likelihood(
                particles, sensor_data['imu']
            )
            weights *= imu_weights

        # Normalize weights
        weights = weights / np.sum(weights)
        return weights

    def get_motion_noise(self):
        """Get motion noise based on humanoid-specific factors"""
        # Higher noise during walking
        # Lower noise when stationary
        # Account for balance adjustments
        noise = np.diag([
            0.01,  # x position noise
            0.01,  # y position noise
            0.005, # z position noise
            0.02,  # roll noise
            0.02,  # pitch noise
            0.03   # yaw noise
        ])
        return noise
```

# Example: Gait-aware mapping for humanoid robots
class GaitAwareMapper:
    def __init__(self):
        self.gait_phase = 0.0
        self.step_count = 0
        self.map_updates = []
        self.gait_frequency = 2.0  # Hz (typical walking frequency)

    def update_map_with_gait_awareness(self, sensor_data, robot_pose, gait_state):
        """Update map considering current gait phase"""
        # Determine if we're in a stable phase of gait
        stable_phase = self.is_gait_phase_stable(gait_state)

        if stable_phase:
            # Perform detailed mapping during stable phases
            map_update = self.create_detailed_map_update(sensor_data, robot_pose)
            self.integrate_map_update(map_update)
        else:
            # Use conservative mapping during unstable phases
            map_update = self.create_conservative_map_update(sensor_data, robot_pose)
            self.integrate_map_update(map_update, conservative=True)

    def is_gait_phase_stable(self, gait_state):
        """Determine if current gait phase is stable for mapping"""
        # Stable phases: double support, stance phase
        # Unstable phases: single support, swing phase
        current_phase = gait_state.phase
        stance_start = gait_state.stance_start
        stance_end = gait_state.stance_end

        # Check if in relatively stable phase
        return stance_start <= current_phase <= stance_end

    def create_detailed_map_update(self, sensor_data, robot_pose):
        """Create detailed map update during stable gait phases"""
        # Use full sensor data for detailed mapping
        detailed_map = self.process_full_sensor_data(sensor_data)

        # Transform to global coordinates
        global_map = self.transform_to_global_coordinates(detailed_map, robot_pose)

        return global_map

    def create_conservative_map_update(self, sensor_data, robot_pose):
        """Create conservative map update during unstable gait phases"""
        # Use only reliable features
        reliable_features = self.extract_reliable_features(sensor_data)

        # Apply more conservative updates
        conservative_map = self.process_reliable_features(reliable_features)

        # Transform to global coordinates
        global_map = self.transform_to_global_coordinates(conservative_map, robot_pose)

        return global_map
